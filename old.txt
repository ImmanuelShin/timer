    // Data model for a single stopwatch
    public class StopwatchData
    {
        public string Name { get; set; } = "";
        public long ElapsedTicks { get; set; }
        public DateTime? StartTime { get; set; }
        public bool IsRunning { get; set; }
        public bool IsPersistent { get; set; }
        public List<LapTime> Laps { get; set; } = new List<LapTime>();
        public DateTime CreatedAt { get; set; }
        public DateTime LastUsed { get; set; }
    }

    // Data model for lap/flag times
    public class LapTime
    {
        public DateTime Timestamp { get; set; }
        public long ElapsedTicks { get; set; }
        public string Note { get; set; } = "";
    }

    // Main persistent stopwatch class
    public class PersistentStopwatch
    {
        private StopwatchData _data;
        private System.Timers.Timer _updateTimer;
        private readonly string _dataFile;

        public event EventHandler<TimeSpan>? TimeUpdated;
        public event EventHandler<LapTime>? LapAdded;

        public string Name => _data.Name;
        public bool IsRunning => _data.IsRunning;
        public TimeSpan Elapsed => GetCurrentElapsed();
        public List<LapTime> Laps => _data.Laps;
        public DateTime CreatedAt => _data.CreatedAt;
        public DateTime LastUsed => _data.LastUsed;

        public PersistentStopwatch(string name, string dataDirectory)
        {
            _dataFile = Path.Combine(dataDirectory, $"{SanitizeFileName(name)}.json");
            _data = LoadData(name);
            
            // Setup update timer (100ms for smooth UI updates)
            _updateTimer = new System.Timers.Timer(100);
            _updateTimer.Elapsed += OnTimerElapsed;
            
            if (_data.IsRunning)
            {
                _updateTimer.Start();
            }
        }


        private void OnTimerElapsed(object? sender, ElapsedEventArgs e)
        {
            if (_data.IsRunning)
            {
                TimeUpdated?.Invoke(this, GetCurrentElapsed());
            }
        }

        private TimeSpan GetCurrentElapsed()
        {
            var totalTicks = _data.ElapsedTicks;
            
            if (_data.IsRunning && _data.StartTime.HasValue)
            {
                totalTicks += (DateTime.Now - _data.StartTime.Value).Ticks;
            }
            
            return new TimeSpan(totalTicks);
        }

        public void Start()
        {
            if (!_data.IsRunning)
            {
                _data.IsRunning = true;
                _data.StartTime = DateTime.Now;
                _data.LastUsed = DateTime.Now;
                _updateTimer.Start();
                SaveData();
            }
        }

        public void Stop()
        {
            if (_data.IsRunning && _data.StartTime.HasValue)
            {
                _data.ElapsedTicks += (DateTime.Now - _data.StartTime.Value).Ticks;
                _data.IsRunning = false;
                _data.StartTime = null;
                _data.LastUsed = DateTime.Now;
                _updateTimer.Stop();
                SaveData();
            }
        }

        public void Reset()
        {
            _data.ElapsedTicks = 0;
            _data.IsRunning = false;
            _data.StartTime = null;
            _data.LastUsed = DateTime.Now;
            _data.Laps.Clear();
            _updateTimer.Stop();
            SaveData();
        }

        public LapTime AddLap(string note = "")
        {
            var lap = new LapTime
            {
                Timestamp = DateTime.Now,
                ElapsedTicks = GetCurrentElapsed().Ticks,
                Note = note
            };
            
            _data.Laps.Add(lap);
            _data.LastUsed = DateTime.Now;
            SaveData();
            
            LapAdded?.Invoke(this, lap);
            return lap;
        }

        private StopwatchData LoadData(string name)
        {
            if (File.Exists(_dataFile))
            {
                try
                {
                    var json = File.ReadAllText(_dataFile);
                    var data = JsonSerializer.Deserialize<StopwatchData>(json);
                    if (data != null)
                    {
                        // Reset running state on load (computer was off)
                        if (data.IsRunning && data.StartTime.HasValue)
                        {
                            // Add the time that passed before shutdown
                            data.ElapsedTicks += (DateTime.Now - data.StartTime.Value).Ticks;
                        }
                        data.IsRunning = false;
                        data.StartTime = null;
                        return data;
                    }
                }
                catch (Exception ex)
                {
                    Console.WriteLine($"Error loading data for {name}: {ex.Message}");
                }
            }

            // Create new stopwatch data
            return new StopwatchData
            {
                Name = name,
                CreatedAt = DateTime.Now,
                LastUsed = DateTime.Now
            };
        }

        private void SaveData()
        {
            try
            {
                var json = JsonSerializer.Serialize(_data, new JsonSerializerOptions 
                { 
                    WriteIndented = true 
                });
                
                Directory.CreateDirectory(Path.GetDirectoryName(_dataFile)!);
                File.WriteAllText(_dataFile, json);
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Error saving data for {_data.Name}: {ex.Message}");
            }
        }

        private static string SanitizeFileName(string name)
        {
            var invalid = Path.GetInvalidFileNameChars();
            return new string(name.Select(c => invalid.Contains(c) ? '_' : c).ToArray());
        }

        public void Dispose()
        {
            _updateTimer?.Dispose();
            if (_data.IsRunning)
            {
                Stop();
            }
        }
    }

    // Manager class for multiple stopwatches
    public class StopwatchManager
    {
        private readonly Dictionary<string, PersistentStopwatch> _stopwatches;
        private readonly string _dataDirectory;

        public IEnumerable<PersistentStopwatch> Stopwatches => _stopwatches.Values;

        public StopwatchManager(string dataDirectory = "StopwatchData")
        {
            _dataDirectory = dataDirectory;
            _stopwatches = new Dictionary<string, PersistentStopwatch>();
            LoadExistingStopwatches();
        }

        private void LoadExistingStopwatches()
        {
            if (!Directory.Exists(_dataDirectory))
                return;

            foreach (var file in Directory.GetFiles(_dataDirectory, "*.json"))
            {
                try
                {
                    var json = File.ReadAllText(file);
                    var data = JsonSerializer.Deserialize<StopwatchData>(json);
                    if (data != null && !string.IsNullOrEmpty(data.Name))
                    {
                        var stopwatch = new PersistentStopwatch(data.Name, _dataDirectory);
                        _stopwatches[data.Name] = stopwatch;
                    }
                }
                catch (Exception ex)
                {
                    Console.WriteLine($"Error loading stopwatch from {file}: {ex.Message}");
                }
            }
        }

        public PersistentStopwatch CreateStopwatch(string name)
        {
            if (string.IsNullOrWhiteSpace(name))
                throw new ArgumentException("Stopwatch name cannot be empty");

            if (_stopwatches.ContainsKey(name))
                throw new ArgumentException($"Stopwatch '{name}' already exists");

            var stopwatch = new PersistentStopwatch(name, _dataDirectory);
            _stopwatches[name] = stopwatch;
            return stopwatch;
        }

        public PersistentStopwatch? GetStopwatch(string name)
        {
            return _stopwatches.TryGetValue(name, out var stopwatch) ? stopwatch : null;
        }

        public bool DeleteStopwatch(string name)
        {
            if (_stopwatches.TryGetValue(name, out var stopwatch))
            {
                stopwatch.Dispose();
                _stopwatches.Remove(name);

                try
                {
                    var dataFile = Path.Combine(_dataDirectory, $"{SanitizeFileName(name)}.json");
                    if (File.Exists(dataFile))
                        File.Delete(dataFile);
                    return true;
                }
                catch (Exception ex)
                {
                    Console.WriteLine($"Error deleting stopwatch file: {ex.Message}");
                }
            }
            return false;
        }

        private static string SanitizeFileName(string name)
        {
            var invalid = Path.GetInvalidFileNameChars();
            return new string(name.Select(c => invalid.Contains(c) ? '_' : c).ToArray());
        }

        public void Dispose()
        {
            foreach (var stopwatch in _stopwatches.Values)
            {
                stopwatch.Dispose();
            }
        }
    }

        private void AllStopwatchesList_ItemActivate(object sender, EventArgs e)
        {
            if (_allStopwatchesList.SelectedItems.Count == 0) return;

            var selectedName = _allStopwatchesList.SelectedItems[0].Text;
            var stopwatch = _manager.GetStopwatch(selectedName);
            if (stopwatch == null) return;

            // set current stopwatch and update the UI immediately
            _currentStopwatch = stopwatch;
            UpdateUI();
            UpdateLapsList();

            // also sync the combobox (only if it contains the item)
            for (int i = 0; i < _stopwatchCombo.Items.Count; i++)
            {
                if (_stopwatchCombo.Items[i]?.ToString() == selectedName)
                {
                    _stopwatchCombo.SelectedIndex = i;
                    break;
                }
            }
        }

        private void UpdateSidebar()
        {
            if (!_sidebarExpanded) return;

            _allStopwatchesList.BeginUpdate();
            try
            {
                // remember selection so we can restore it
                string? selectedName = null;
                if (_allStopwatchesList.SelectedItems.Count > 0)
                    selectedName = _allStopwatchesList.SelectedItems[0].Text;

                // map existing items by name
                var existing = _allStopwatchesList.Items
                    .Cast<ListViewItem>()
                    .ToDictionary(i => i.Text, i => i);

                var seen = new HashSet<string>();

                foreach (var stopwatch in _manager.Stopwatches.OrderBy(s => s.Name))
                {
                    seen.Add(stopwatch.Name);

                    if (!existing.TryGetValue(stopwatch.Name, out var item))
                    {
                        // create item with required subitems
                        item = new ListViewItem(stopwatch.Name);
                        item.SubItems.Add(""); // Duration
                        item.SubItems.Add(""); // Status
                        item.SubItems.Add(""); // Last Used
                        _allStopwatchesList.Items.Add(item);
                    }

                    // Update subitems in-place
                    var elapsed = stopwatch.Elapsed;
                    item.SubItems[1].Text = $"{elapsed:hh\\:mm\\:ss}";
                    item.SubItems[2].Text = stopwatch.IsRunning ? "RUNNING" : "STOPPED";

                    var timeAgo = DateTime.Now - stopwatch.LastUsed;
                    string timeAgoText;
                    if (timeAgo.TotalMinutes < 1)
                        timeAgoText = "Now";
                    else if (timeAgo.TotalHours < 1)
                        timeAgoText = $"{(int)timeAgo.TotalMinutes}m ago";
                    else if (timeAgo.TotalDays < 1)
                        timeAgoText = $"{(int)timeAgo.TotalHours}h ago";
                    else
                        timeAgoText = $"{(int)timeAgo.TotalDays}d ago";

                    item.SubItems[3].Text = timeAgoText;

                    // Colors: prefer highlighting selected stopwatch above running state
                    if (_currentStopwatch?.Name == stopwatch.Name)
                        item.BackColor = Color.LightBlue;
                    else if (stopwatch.IsRunning)
                        item.BackColor = Color.LightGreen;
                    else
                        item.BackColor = Color.White;
                }

                // Remove items that no longer exist
                for (int i = _allStopwatchesList.Items.Count - 1; i >= 0; i--)
                {
                    var it = _allStopwatchesList.Items[i];
                    if (!seen.Contains(it.Text))
                        _allStopwatchesList.Items.RemoveAt(i);
                }

                // restore selection if still present
                if (!string.IsNullOrEmpty(selectedName))
                {
                    var restore = _allStopwatchesList.Items.Cast<ListViewItem>().FirstOrDefault(x => x.Text == selectedName);
                    if (restore != null)
                        restore.Selected = true;
                }
            }
            finally
            {
                _allStopwatchesList.EndUpdate();
            }
        }

        private void LoadStopwatches()
        {
            _stopwatchCombo.Items.Clear();
            foreach (var sw in _manager.Stopwatches.OrderBy(s => s.Name))
            {
                _stopwatchCombo.Items.Add(sw.Name);
            }

            if (_stopwatchCombo.Items.Count > 0 && _currentStopwatch == null)
            {
                _stopwatchCombo.SelectedIndex = 0;
            }
            else if (_currentStopwatch != null)
            {
                _stopwatchCombo.SelectedItem = _currentStopwatch.Name;
            }

            UpdateUI();
            UpdateSidebar();
        }

        private void StopwatchCombo_SelectedIndexChanged(object sender, EventArgs e)
        {
            if (_stopwatchCombo.SelectedItem != null)
            {
                _currentStopwatch = _manager.GetStopwatch(_stopwatchCombo.SelectedItem.ToString()!);
                UpdateUI();
                UpdateLapsList();
            }
        }

                }

        {

                    MessageBoxButtons.YesNo, MessageBoxIcon.Question);

                    _manager.DeleteStopwatch(_currentStopwatch.Name);

                    UpdateSidebar();
                }
            }
        }

        private void StartButton_Click(object sender, EventArgs e)

        }

        private void LapButton_Click(object sender, EventArgs e)
        {
            if (_currentStopwatch != null)
            {
                _currentStopwatch.AddLap(_lapNoteText.Text.Trim());
                _lapNoteText.Clear();
                UpdateLapsList();
            }
        }

        private void UiTimer_Tick(object sender, EventArgs e)
        {
            if (_currentStopwatch?.IsRunning == true)
            {
                UpdateTimeDisplay();
            }

            // update sidebar less frequently to avoid interfering with clicks
            _sidebarUpdateCounter++;
            if (_sidebarUpdateCounter >= 5) // 5 * 100ms = 500ms
            {
                _sidebarUpdateCounter = 0;
                UpdateSidebar();
            }
        }
        private void UpdateUI()
        {
            var hasStopwatch = _currentStopwatch != null;
            var isRunning = _currentStopwatch?.IsRunning == true;

            _startButton.Enabled = hasStopwatch && !isRunning;
            _stopButton.Enabled = hasStopwatch && isRunning;
            _resetButton.Enabled = hasStopwatch;
            _lapButton.Enabled = hasStopwatch;
            _lapNoteText.Enabled = hasStopwatch;
            _deleteButton.Enabled = hasStopwatch;

            UpdateTimeDisplay();
        }

        private void UpdateTimeDisplay()
        {
            if (_currentStopwatch != null)
            {
                var elapsed = _currentStopwatch.Elapsed;
                _timeLabel.Text = $"{elapsed:hh\\:mm\\:ss\\.fff}";
                _timeLabel.ForeColor = _currentStopwatch.IsRunning ? Color.Green : Color.Black;
            }
            else
            {
                _timeLabel.Text = "00:00:00.000";
                _timeLabel.ForeColor = Color.Gray;
            }
        }

        private void UpdateLapsList()
        {
            _lapsList.Items.Clear();
            if (_currentStopwatch?.Laps.Any() == true)
            {
                foreach (var lap in _currentStopwatch.Laps)
                {
                    var elapsed = new TimeSpan(lap.ElapsedTicks);
                    var item = $"{elapsed:hh\\:mm\\:ss} - {lap.Timestamp:HH:mm:ss}";
                    if (!string.IsNullOrWhiteSpace(lap.Note))
                        item += $" ({lap.Note})";
                    _lapsList.Items.Add(item);
                }
            }
        }

        protected override void OnFormClosing(FormClosingEventArgs e)
        {
            _uiTimer?.Stop();
            _manager?.Dispose();
            base.OnFormClosing(e);
        }
